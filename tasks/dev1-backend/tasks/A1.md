## Skeleton: `/api/main.py`

```python
from contextlib import asynccontextmanager
from fastapi import FastAPI

from api.config import settings
from api.db import engine, init_db, close_db
from api.redis import redis_client, close_redis
from api.routes import healthz, company_profiles, agent_runs, opportunities, action_packages


@asynccontextmanager
async def lifespan(app: FastAPI):
    await init_db()
    # Mock SSE replay registers here until A9 ships
    yield
    await close_db()
    await close_redis()


app = FastAPI(title="GovCapture API", version="0.1.0", lifespan=lifespan)

app.include_router(healthz.router)
app.include_router(company_profiles.router)
app.include_router(agent_runs.router)
app.include_router(opportunities.router)
app.include_router(action_packages.router)
```

## Skeleton: `/api/config.py`

```python
from functools import lru_cache
from pydantic import Field
from pydantic_settings import BaseSettings, SettingsConfigDict


class Settings(BaseSettings):
    model_config = SettingsConfigDict(env_file=".env", extra="ignore")

    anthropic_api_key: str = Field(..., alias="ANTHROPIC_API_KEY")
    sam_api_key: str = Field("", alias="SAM_API_KEY")
    database_url: str = Field(..., alias="DATABASE_URL")
    redis_url: str = Field(..., alias="REDIS_URL")
    hermes_home: str = Field("/var/lib/hermes", alias="HERMES_HOME")
    hermes_model: str = Field("claude-sonnet-4-6", alias="HERMES_MODEL")

    llm_dev_model: str = Field("claude-haiku-4-5-20251001", alias="LLM_DEV_MODEL")
    llm_synth_model: str = Field("claude-sonnet-4-6", alias="LLM_SYNTH_MODEL")

    run_budget_usd: float = Field(0.50, alias="RUN_BUDGET_USD")
    run_budget_steps: int = Field(40, alias="RUN_BUDGET_STEPS")
    run_budget_seconds: int = Field(360, alias="RUN_BUDGET_SECONDS")

    demo_use_seeded_only: bool = Field(False, alias="DEMO_USE_SEEDED_ONLY")


@lru_cache
def get_settings() -> Settings:
    return Settings()  # raises on missing required vars at startup


settings = get_settings()
```

This is the **only** file allowed to read `os.environ`. Everywhere else imports `settings`.

## Skeleton: `/api/db.py`

```python
from sqlalchemy.ext.asyncio import AsyncEngine, async_sessionmaker, create_async_engine

from api.config import settings


engine: AsyncEngine = create_async_engine(settings.database_url, echo=False, pool_pre_ping=True)
SessionLocal = async_sessionmaker(engine, expire_on_commit=False)


async def init_db() -> None:
    # Light readiness check — actual schema is owned by Alembic (A2)
    async with engine.connect() as conn:
        await conn.execute(text("select 1"))


async def close_db() -> None:
    await engine.dispose()
```

## Skeleton: `/api/redis.py`

```python
import redis.asyncio as aioredis

from api.config import settings


redis_client = aioredis.from_url(settings.redis_url, decode_responses=True)


async def close_redis() -> None:
    await redis_client.close()
```

## Skeleton: `/api/routes/healthz.py`

```python
from fastapi import APIRouter
from sqlalchemy import text

from api.db import engine
from api.redis import redis_client


router = APIRouter(tags=["health"])


@router.get("/healthz")
async def healthz() -> dict[str, str]:
    # Light dependency check — fail fast if DB or Redis is wedged
    async with engine.connect() as conn:
        await conn.execute(text("select 1"))
    await redis_client.ping()
    return {"status": "ok"}
```

## Skeleton: `/docker-compose.yaml`

```yaml
services:
  api:
    build: ./api
    ports:
      - "8000:8000"
    environment:
      DATABASE_URL: postgresql+asyncpg://govcon:govcon@postgres:5432/govcon
      REDIS_URL: redis://redis:6379/0
    env_file:
      - .env
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    volumes:
      - ./api:/app/api
      - ./schemas:/app/schemas:ro
      - ./fixtures:/app/fixtures:ro
    command: ["uvicorn", "api.main:app", "--host", "0.0.0.0", "--port", "8000", "--reload"]

  postgres:
    image: postgres:16
    environment:
      POSTGRES_USER: govcon
      POSTGRES_PASSWORD: govcon
      POSTGRES_DB: govcon
    volumes:
      - pg_data:/var/lib/postgresql/data
    ports:
      - "5432:5432"
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U govcon"]
      interval: 5s
      timeout: 5s
      retries: 10

  redis:
    image: redis:7
    command: ["redis-server", "--appendonly", "yes"]
    volumes:
      - redis_data:/data
    ports:
      - "6379:6379"
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      timeout: 3s
      retries: 5

  # Hermes Agent: installed in the api container during build (see Dockerfile)
  # No separate service needed — Hermes runs as a subprocess invoked from FastAPI (see A9).

volumes:
  pg_data:
  redis_data:
```

## Pitfalls

- **`asyncpg` vs `psycopg`:** if you pick `asyncpg`, the URL prefix is `postgresql+asyncpg://`. Don't mix them.
- **Volume permissions:** macOS Docker can be slow on bind-mounted Python source. If `--reload` is sluggish, switch to `watchgod` or accept the latency.
- **Forgetting healthcheck `depends_on: condition: service_healthy`:** the API will start before Postgres is ready and crash. Always use the condition.
- **Env vars typed as the wrong type:** Pydantic settings will silently coerce. Test that booleans are actually booleans (`DEMO_USE_SEEDED_ONLY=false` parses to `False`, not the string `"false"`).

## Hermes install in the api container

Add to `/api/Dockerfile`:

```dockerfile
FROM python:3.12-slim

# System deps for Hermes + headless browser tools (Playwright via Hermes built-ins)
RUN apt-get update && apt-get install -y --no-install-recommends \
    curl ca-certificates git build-essential \
    && rm -rf /var/lib/apt/lists/*

# uv (fast Python package manager, used by Hermes)
RUN curl -LsSf https://astral.sh/uv/install.sh | sh
ENV PATH="/root/.local/bin:$PATH"

# Hermes Agent
RUN curl -fsSL https://raw.githubusercontent.com/NousResearch/hermes-agent/main/scripts/install.sh | bash

# Project deps
WORKDIR /app
COPY api/pyproject.toml api/uv.lock ./api/
RUN cd api && uv sync --frozen

# Project source (overlaid by bind mount in dev compose)
COPY api ./api
COPY schemas ./schemas
COPY fixtures ./fixtures

EXPOSE 8000
CMD ["uvicorn", "api.main:app", "--host", "0.0.0.0", "--port", "8000"]
```

Smoke-test after `make dev`:

```bash
docker compose exec api hermes --version
docker compose exec api hermes model    # confirms which model is active
```
