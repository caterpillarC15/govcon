## Skeleton: `routes/agent_runs.py`

```python
import asyncio
import json
from uuid import UUID
from fastapi import APIRouter, Depends, HTTPException
from fastapi.responses import StreamingResponse

from api.schemas.agent_run import AgentRun, AgentRunCreate
from api.repositories.agent_run_repo import AgentRunRepository
from api.deps import get_agent_run_repo
from api.redis import redis_client


router = APIRouter(prefix="/agent-runs", tags=["agent-runs"])


@router.post("", response_model=AgentRun, status_code=201)
async def create_agent_run(
    payload: AgentRunCreate,
    repo: AgentRunRepository = Depends(get_agent_run_repo),
) -> AgentRun:
    run = await repo.create(payload)
    # A9 will trigger the planner via a background task or message queue here.
    # For now, emit a pre-recorded trace if DEMO_USE_SEEDED_ONLY=true and no planner exists yet.
    return run


@router.get("/{run_id}", response_model=AgentRun)
async def get_agent_run(run_id: UUID, repo: AgentRunRepository = Depends(get_agent_run_repo)) -> AgentRun:
    run = await repo.get(run_id)
    if not run:
        raise HTTPException(404, "Agent run not found")
    return run


@router.get("/{run_id}/stream")
async def stream_agent_run(run_id: UUID) -> StreamingResponse:
    async def event_gen():
        # Real path: subscribe to Redis pub/sub channel `agent-run:<run_id>`
        # Until A9 ships, replay /schemas/trace-event.example.jsonl at captured cadence.
        pubsub = redis_client.pubsub()
        await pubsub.subscribe(f"agent-run:{run_id}")
        try:
            async for message in pubsub.listen():
                if message["type"] != "message":
                    continue
                event = json.loads(message["data"])
                yield f"event: {event['type']}\ndata: {json.dumps(event)}\n\n"
                if event["type"] == "run_completed":
                    break
        finally:
            await pubsub.unsubscribe(f"agent-run:{run_id}")

    return StreamingResponse(
        event_gen(),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"},
    )
```

## Skeleton: `repositories/agent_run_repo.py`

from uuid import UUID
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from api.db.models import AgentRun as AgentRunModel
from api.schemas.agent_run import AgentRun, AgentRunCreate


class AgentRunRepository:
    def __init__(self, session: AsyncSession) -> None:
        self.session = session

    async def create(self, payload: AgentRunCreate) -> AgentRun:
        row = AgentRunModel(
            goal=payload.goal,
            company_profile_id=payload.profile_id,
            status="pending",
            steps=[],
            opportunities=[],
        )
        self.session.add(row)
        await self.session.commit()
        await self.session.refresh(row)
        return AgentRun.model_validate(row, from_attributes=True)

    async def get(self, run_id: UUID) -> AgentRun | None:
        row = await self.session.get(AgentRunModel, run_id)
        return AgentRun.model_validate(row, from_attributes=True) if row else None

## SSE framing rules

CONTRACTS.md §3 is canonical. To prevent regressions:

1. Every event line: `event: <type>\ndata: <json>\n\n` — note the **double newline at the end**, single newline between `event:` and `data:`.
2. Content type **must** be `text/event-stream`.
3. Send a `:keepalive\n\n` comment every 15 seconds if the run is silent for that long. Some proxies time out idle SSE.
4. On `run_completed`, close the stream cleanly.

## Pitfalls

- **Forgetting `X-Accel-Buffering: no` header on SSE:** nginx (in prod) buffers responses by default; the stream appears to drop until completion. Set the header on the response.
- **Returning ORM objects directly:** always go through Pydantic models. Validates the response shape against the contract.
- **Pagination:** for v1 you can skip — fixtures are small. Add `?limit=&offset=` only if Dev 2 asks.
- **Auth:** PRD §17 Q5 defaults to none for MVP. Don't add auth unless the user explicitly requests it.
- **CORS:** the web app on Vercel hits the API on VX1. Add `CORSMiddleware` allowing the Vercel origin and localhost:3000.
