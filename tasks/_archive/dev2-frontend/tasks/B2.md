# B2 — API client + zod-validated types

**Goal:** A typed API client that fetches every CONTRACTS.md §6 endpoint and validates responses against generated zod schemas. SSE consumer wraps `EventSource` and emits a typed `TraceEvent` discriminated union.

## Skeleton: lib/api.ts

```ts
import { z } from "zod";
import { env } from "@/lib/env";
import {
  CompanyProfileSchema, AgentRunSchema, OpportunitySchema,
  ExtractedRequirementSchema, FitScoreSchema, RiskFlagSchema, ActionPackageSchema,
} from "@/lib/schemas";

const BASE = env.NEXT_PUBLIC_API_BASE;

class ApiError extends Error {
  constructor(public status: number, message: string, public body?: unknown) {
    super(message);
  }
}

async function jget<T>(path: string, schema: z.ZodSchema<T>, init?: RequestInit): Promise<T> {
  const res = await fetch(`${BASE}${path}`, { cache: "no-store", ...init });
  if (!res.ok) {
    const body = await res.text().catch(() => "");
    throw new ApiError(res.status, `${res.status} ${res.statusText} ${path}`, body);
  }
  const data = await res.json();
  const parsed = schema.safeParse(data);
  if (!parsed.success) {
    if (process.env.NODE_ENV !== "production") {
      console.error("Zod validation failed for", path, parsed.error.format(), data);
    }
    throw new ApiError(200, `response shape mismatch on ${path}`, data);
  }
  return parsed.data;
}

async function jpost<T, U>(
  path: string, body: unknown, schema: z.ZodSchema<T>
): Promise<T> {
  return jget(path, schema, {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify(body),
  });
}

export const api = {
  // Profiles
  createProfile: (body: unknown) => jpost("/company-profiles", body, CompanyProfileSchema),
  getProfile: (id: string) => jget(`/company-profiles/${id}`, CompanyProfileSchema),

  // Agent runs
  createRun: (body: { profile_id: string; goal: string }) =>
    jpost("/agent-runs", body, AgentRunSchema),
  getRun: (id: string) => jget(`/agent-runs/${id}`, AgentRunSchema),
  getRunOpportunities: (id: string) =>
    jget(`/agent-runs/${id}/opportunities`, z.array(OpportunitySchema)),

  // Opportunity detail
  getOpportunity: (id: string) => jget(`/opportunities/${id}`, OpportunitySchema),
  getRequirements: (id: string) =>
    jget(`/opportunities/${id}/requirements`, z.array(ExtractedRequirementSchema)),
  getFitScore: (id: string) => jget(`/opportunities/${id}/fit-score`, FitScoreSchema),
  getRisks: (id: string) => jget(`/opportunities/${id}/risks`, z.array(RiskFlagSchema)),

  // Action package
  getActionPackage: (id: string) => jget(`/action-packages/${id}`, ActionPackageSchema),

  // Health
  healthz: () => jget("/healthz", z.object({ status: z.string() })),
};
```

## Skeleton: lib/sse.ts

```ts
import { TraceEventSchema, TraceEvent } from "@/lib/schemas";

export type SseHandlers = {
  onEvent: (event: TraceEvent) => void;
  onError?: (err: Event) => void;
  onOpen?: () => void;
};

export function streamAgentRun(runId: string, handlers: SseHandlers): () => void {
  const url = `${env.NEXT_PUBLIC_API_BASE}/agent-runs/${runId}/stream`;
  const es = new EventSource(url);

  if (handlers.onOpen) es.onopen = handlers.onOpen;
  if (handlers.onError) es.onerror = handlers.onError;

  // Each trace-event type is emitted with `event: <type>` framing — listen per type.
  // (Standard EventSource fires `message` only for unnamed events.)
  for (const type of TraceEventTypes) {
    es.addEventListener(type, (msg: MessageEvent) => {
      try {
        const data = JSON.parse(msg.data);
        const parsed = TraceEventSchema.safeParse(data);
        if (parsed.success) {
          handlers.onEvent(parsed.data);
        } else if (process.env.NODE_ENV !== "production") {
          console.warn("Invalid trace event shape:", parsed.error, data);
        }
      } catch (e) {
        console.warn("Failed to parse trace event:", e, msg.data);
      }
    });
  }

  return () => es.close();
}

const TraceEventTypes = [
  "run_started", "step_started", "tool_called", "tool_returned",
  "step_completed", "opportunity_ranked", "needs_human", "run_completed",
] as const;
```

## Skeleton: hooks/useAgentRunStream.ts

```ts
import { useEffect, useReducer } from "react";
import { streamAgentRun } from "@/lib/sse";
import type { TraceEvent } from "@/lib/schemas";

type State = {
  events: TraceEvent[];
  status: "connecting" | "streaming" | "completed" | "error";
  rankedOpportunities: { id: string; score: number; decision: string }[];
};

type Action =
  | { type: "open" }
  | { type: "event"; event: TraceEvent }
  | { type: "error" };

function reducer(state: State, action: Action): State {
  switch (action.type) {
    case "open":
      return { ...state, status: "streaming" };
    case "error":
      return { ...state, status: "error" };
    case "event": {
      const events = [...state.events, action.event];
      let ranked = state.rankedOpportunities;
      let status = state.status;
      if (action.event.type === "opportunity_ranked") {
        ranked = [...ranked, {
          id: action.event.opportunity_id,
          score: action.event.score,
          decision: action.event.decision,
        }];
      }
      if (action.event.type === "run_completed") status = "completed";
      return { ...state, events, rankedOpportunities: ranked, status };
    }
  }
}

export function useAgentRunStream(runId: string) {
  const [state, dispatch] = useReducer(reducer, {
    events: [],
    status: "connecting",
    rankedOpportunities: [],
  });

  useEffect(() => {
    if (!runId) return;
    const close = streamAgentRun(runId, {
      onOpen: () => dispatch({ type: "open" }),
      onEvent: (event) => dispatch({ type: "event", event }),
      onError: () => dispatch({ type: "error" }),
    });
    return close;
  }, [runId]);

  return state;
}
```

## React Query for static endpoints

Wrap the rest with `@tanstack/react-query` for caching and re-fetching:

```bash
npm install @tanstack/react-query
```

```tsx
// app/providers.tsx
'use client';
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { useState } from "react";

export function Providers({ children }) {
  const [client] = useState(() => new QueryClient({
    defaultOptions: { queries: { staleTime: 30_000, retry: 1 } },
  }));
  return <QueryClientProvider client={client}>{children}</QueryClientProvider>;
}
```

Then per-endpoint hooks:

```ts
import { useQuery } from "@tanstack/react-query";
import { api } from "@/lib/api";

export function useOpportunity(id: string) {
  return useQuery({
    queryKey: ["opportunity", id],
    queryFn: () => api.getOpportunity(id),
    enabled: !!id,
  });
}
```

## Pitfalls

- **`event:`-named SSE events vs default `message`:** `EventSource.onmessage` only fires for events without a `event:` line. We use `addEventListener('<type>', ...)` per type. Get this wrong and you see nothing.
- **Stale generated schemas:** if A regenerates schemas and pushes, your `tsc` breaks until you `make schemas` locally. Add a pre-commit hook or check in CI.
- **Caching SSE responses:** never. Always `cache: "no-store"`.
- **Reading process.env outside `lib/env.ts`:** flag it in lint. Centralized env reading is what keeps the contract sane.
- **Silent zod errors in production:** the API client throws on validation failure but production users shouldn't see a parse error. Show a friendly retry UI; log to a backend later if you add observability.
