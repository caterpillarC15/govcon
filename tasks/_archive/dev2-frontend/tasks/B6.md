## OpportunityCard.tsx

```tsx
import Link from "next/link";
import { DecisionBadge } from "@/components/ui/DecisionBadge";
import type { Opportunity, FitScore, RiskFlag } from "@/lib/schemas";

export function OpportunityCard({
  opportunity, fitScore, topRisk,
}: {
  opportunity: Opportunity;
  fitScore?: FitScore;
  topRisk?: RiskFlag;
}) {
  const dueIn = opportunity.due_date
    ? Math.ceil((new Date(opportunity.due_date).getTime() - Date.now()) / 86_400_000)
    : null;

  return (
    <Link
      href={`/opportunities/${opportunity.id}`}
      className="block rounded-lg border border-canvas-border bg-canvas p-6 hover:border-accent transition-colors"
    >
      <div className="flex items-start justify-between gap-3">
        <div className="min-w-0">
          <h3 className="text-base font-semibold text-ink truncate">{opportunity.title}</h3>
          <p className="mt-1 text-xs text-ink-muted">
            {opportunity.agency} · {opportunity.solicitation_number}
          </p>
        </div>
        {fitScore && <DecisionBadge decision={fitScore.decision} score={fitScore.total_score} />}
      </div>

      <dl className="mt-4 grid grid-cols-2 gap-x-4 gap-y-2 text-xs">
        <div>
          <dt className="text-ink-muted">NAICS</dt>
          <dd className="text-ink">{opportunity.naics ?? "—"}</dd>
        </div>
        <div>
          <dt className="text-ink-muted">Set-aside</dt>
          <dd className="text-ink">{opportunity.set_aside ?? "—"}</dd>
        </div>
        <div>
          <dt className="text-ink-muted">Due</dt>
          <dd className="text-ink">
            {opportunity.due_date ?? "—"}
            {dueIn !== null && <span className="ml-1 text-ink-muted">({dueIn}d)</span>}
          </dd>
        </div>
        <div>
          <dt className="text-ink-muted">Place</dt>
          <dd className="text-ink truncate">{opportunity.place_of_performance ?? "—"}</dd>
        </div>
      </dl>

      {fitScore?.recommended_next_action && (
        <p className="mt-4 text-sm text-ink leading-6">
          <strong className="text-ink-muted">Top reason:</strong> {fitScore.recommended_next_action}
        </p>
      )}

      {topRisk && (
        <p className="mt-2 text-sm text-ink-muted leading-6">
          <strong className="text-decision-reject">Main risk:</strong> {topRisk.title}
        </p>
      )}
    </Link>
  );
}
```

## RankedList.tsx

Wraps cards in a grid with sort + filter:

```tsx
import { useMemo, useState } from "react";
import { OpportunityCard } from "./OpportunityCard";

type Bundle = { opportunity: Opportunity; fitScore?: FitScore; topRisk?: RiskFlag };

export function RankedList({ items }: { items: Bundle[] }) {
  const [decisionFilter, setDecisionFilter] = useState<"all" | "strong_pursue" | "pursue" | "maybe" | "reject">("all");

  const filtered = useMemo(() =>
    items
      .filter(b => decisionFilter === "all" || b.fitScore?.decision === decisionFilter)
      .sort((a, b) => (b.fitScore?.total_score ?? 0) - (a.fitScore?.total_score ?? 0))
  , [items, decisionFilter]);

  return (
    <div>
      <div className="mb-4 flex flex-wrap gap-2">
        {(["all", "strong_pursue", "pursue", "maybe", "reject"] as const).map(d => (
          <button
            key={d}
            className={`rounded-full border px-3 py-1 text-xs ${
              decisionFilter === d
                ? "border-accent bg-accent text-canvas"
                : "border-canvas-border bg-canvas text-ink hover:border-accent"
            }`}
            onClick={() => setDecisionFilter(d)}
          >
            {d.replace("_", " ")}
          </button>
        ))}
      </div>

      {filtered.length === 0 ? (
        <p className="text-sm text-ink-muted">No opportunities match this filter.</p>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
          {filtered.map(b => <OpportunityCard key={b.opportunity.id} {...b} />)}
        </div>
      )}
    </div>
  );
}
```

## Pitfalls

- **Fetching fit-score per opportunity = N+1.** Add a `GET /agent-runs/:id/opportunities-with-scores` endpoint if it gets slow. v1 with 3-5 opportunities is fine.
- **Card size inconsistency.** When titles vary in length, cards lose vertical rhythm. Either truncate at 2 lines (`line-clamp-2`) or set min-height.
- **Filter count badge.** Optional but nice: show `(3)` next to the active filter pill.
- **Decision-band spelling.** Match the API exactly: `strong_pursue` (underscore). Don't transform to `strongPursue` in UI text.
