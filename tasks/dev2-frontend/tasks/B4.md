## Skeleton: app/runs/new/page.tsx

```tsx
'use client';

import { Suspense } from "react";
import { useRouter, useSearchParams } from "next/navigation";
import { useState } from "react";
import { useCompanyProfile } from "@/lib/hooks/useCompanyProfile";
import { GoalInput } from "@/components/runs/GoalInput";
import { SuggestedGoals } from "@/components/runs/SuggestedGoals";
import { Button } from "@/components/ui/button";
import { api } from "@/lib/api";
import { toast } from "sonner";

export default function NewRunPage() {
  return (
    <Suspense>
      <NewRunInner />
    </Suspense>
  );
}

function NewRunInner() {
  const router = useRouter();
  const params = useSearchParams();
  const profileId = params.get("profile");
  const { data: profile, isLoading } = useCompanyProfile(profileId);
  const [goal, setGoal] = useState("");
  const [submitting, setSubmitting] = useState(false);

  if (!profileId) {
    return (
      <main className="mx-auto max-w-2xl px-6 py-12 text-center">
        <p className="text-ink-muted">Pick a profile first.</p>
        <Button asChild className="mt-4"><a href="/profile/new">Create profile</a></Button>
      </main>
    );
  }

  if (isLoading || !profile) return <main className="px-6 py-12">Loading…</main>;

  return (
    <main className="mx-auto max-w-2xl px-6 py-12">
      <h1 className="text-3xl font-semibold tracking-tight text-ink">What are you looking for?</h1>
      <p className="mt-2 text-sm text-ink-muted">
        Describe in plain language. The agent parses it into search criteria.
      </p>

      <div className="mt-6 rounded-md border border-canvas-border bg-canvas-subtle p-4 text-sm">
        <strong>Profile:</strong> {profile.name}
        <span className="text-ink-muted"> · {profile.capabilities?.slice(0, 3).join(", ")}</span>
      </div>

      <div className="mt-6">
        <GoalInput value={goal} onChange={setGoal} />
      </div>

      <div className="mt-4">
        <p className="text-xs uppercase tracking-wider text-ink-muted">Try one of these</p>
        <SuggestedGoals onPick={setGoal} profile={profile} />
      </div>

      <div className="mt-8 flex justify-end">
        <Button
          disabled={submitting || goal.trim().length < 20}
          onClick={async () => {
            setSubmitting(true);
            try {
              const run = await api.createRun({ profile_id: profileId, goal: goal.trim() });
              router.push(`/runs/${run.id}`);
            } catch (e) {
              toast.error("Failed to start run.");
              setSubmitting(false);
            }
          }}
        >
          Start capture run
        </Button>
      </div>
    </main>
  );
}
```

## SuggestedGoals.tsx

Per PRD §5.2 examples + customized to profile capabilities:

```tsx
const TEMPLATES = [
  "Find {capability} opportunities we can pursue in the next 60 days.",
  "Find facilities maintenance contracts near {location}.",
  "Find {capability} opportunities where we can subcontract.",
  "Find small-business set-aside contracts that match our capabilities.",
  "Find federal contracts under $500K matching our profile.",
];

export function SuggestedGoals({ profile, onPick }: { profile: CompanyProfile; onPick: (g: string) => void }) {
  const cap = profile.capabilities?.[0] ?? "cybersecurity";
  const loc = profile.location ?? "Texas";
  const goals = TEMPLATES.map(t => t.replace("{capability}", cap).replace("{location}", loc));

  return (
    <div className="mt-2 flex flex-wrap gap-2">
      {goals.map((g, i) => (
        <button
          key={i}
          className="rounded-full border border-canvas-border bg-canvas px-3 py-1 text-xs text-ink hover:border-accent hover:text-accent transition-colors"
          onClick={() => onPick(g)}
        >
          {g}
        </button>
      ))}
    </div>
  );
}
```

## GoalInput.tsx

```tsx
import { Textarea } from "@/components/ui/textarea";

export function GoalInput({ value, onChange }: { value: string; onChange: (v: string) => void }) {
  return (
    <div>
      <label className="text-sm font-medium text-ink" htmlFor="goal">Goal</label>
      <Textarea
        id="goal"
        value={value}
        onChange={(e) => onChange(e.target.value)}
        rows={3}
        placeholder="Find cybersecurity opportunities we can pursue in the next 60 days."
        className="mt-1"
      />
      <p className="mt-1 text-xs text-ink-muted">
        Minimum 20 characters. {value.trim().length}/20
      </p>
    </div>
  );
}
```

## Done when

- [ ] User can pick a profile, type or click a suggested goal, and click "Start capture run".
- [ ] Submit calls `POST /agent-runs` and redirects to `/runs/[id]`.
- [ ] Suggested goals are profile-aware (use profile capability and location for substitution).
- [ ] Submit button disabled until goal ≥ 20 chars.
- [ ] No profile in URL → friendly "pick a profile first" view.

## Verify

Manual flow: profile → goal → start → land on `/runs/<id>` with the SSE timeline live.

## Pitfalls

- **Goal too short.** Not enough text and the agent's goal parsing produces nothing useful. Hard min at 20.
- **Demo timing.** When demoing, type fast — the SuggestedGoals chip click is faster but feels canned. Decide which way you like for the stage.
- **Profile-aware substitution miss.** If profile has empty `capabilities`, fall back to "cybersecurity" or whatever's reasonable; don't substitute "{capability}" literally.
